\newcommand{\letterNameTwoTone}[1]{%
  \begingroup
  \def\FirstName{}%
  \def\RemainingName{}%
  \expandafter\letterNameSplit#1 \relax
  \ifstrempty{\RemainingName}{%
    \textcolor{namedark}{\MakeUppercase{#1}}%
  }{%
    \textcolor{namedark}{\MakeUppercase{\FirstName}}\space
    \textcolor{namelight}{\MakeUppercase{\RemainingName}}%
  }%
  \endgroup
}

\def\letterNameSplit#1 #2\relax{%
  \def\FirstName{#1}%
  \def\RemainingName{#2}%
}

% JobOps: #1 is the first contact line, #2 the second (either may be empty),
% #3 the link text, #4 the full link target, #5 the location.
\newcommand{\letterContactBlock}[5]{%
  \begingroup
  \setlength{\parskip}{0pt}%
  \setlength{\parindent}{0pt}%
  \fontsize{8.0}{8.8}\selectfont
  {\bfseries\titlefont{\fontsize{9.2}{9.8}\selectfont CONTACT}}\par
  \ifstrempty{#1}{}{#1\par}%
  \ifstrempty{#2}{}{#2\par}%
  \ifstrempty{#3}{}{{\href{#4}{#3}}\par}%
  \ifstrempty{#5}{}{#5\par}%
  \endgroup
}

\newcommand{\letterHeader}[6]{%
  \begin{flushright}
    \letterContactBlock{#3}{#4}{#5}{#6}{#2}
  \end{flushright}
  \vspace{14pt}%
  {\centering\fontsize{28}{32}\selectfont\bfseries\titlefont{\letterNameTwoTone{#1}}\par}
  \vspace{18pt}%
}
