\newcommand{\resumeNameTwoTone}[1]{%
  \begingroup
  \def\FirstName{}%
  \def\RemainingName{}%
  \expandafter\resumeNameSplit#1 \relax
  \ifstrempty{\RemainingName}{%
    \textcolor{namedark}{\MakeUppercase{#1}}%
  }{%
    \textcolor{namedark}{\MakeUppercase{\FirstName}}\space
    \textcolor{namelight}{\MakeUppercase{\RemainingName}}%
  }%
  \endgroup
}

\def\resumeNameSplit#1 #2\relax{%
  \def\FirstName{#1}%
  \def\RemainingName{#2}%
}

\newcommand{\resumeFullWidthBox}[1]{%
  \noindent\makebox[0pt][l]{%
    \hspace*{-\dimexpr 1in+\hoffset+\oddsidemargin\relax}#1%
  }\par
}

\newcommand{\resumeSection}[1]{%
  \Needspace{5\baselineskip}%
  \vspace{14pt}%
  {\fontsize{11}{13}\selectfont\bfseries\color{resumeaccent}\headerfont{\MakeUppercase{#1}}}\par
  \vspace{4pt}%
  \global\firstsubheadinginsectiontrue
}

% JobOps: #2 (location) and #4 (link) may be empty; #4 is shown without https://
% and #5 is the full link target. #6 is the optional tailored headline.
\newcommand{\resumeHeader}[6]{%
  {\centering\fontsize{26}{30}\selectfont\bfseries\titlefont{\resumeNameTwoTone{#1}}\par}
  \ifstrempty{#6}{}{%
    \vspace{4pt}%
    {\centering\small\color{namedark}\headerfont{#6}\par}%
  }%
  \vspace{14pt}%
  \resumeFullWidthBox{%
    \colorbox{contactbg}{%
      \parbox{\dimexpr\paperwidth-2\fboxsep\relax}{%
        \centering\vspace{4pt}\small #3%
        \ifstrempty{#2}{}{\enspace\enspace\textbar\enspace\enspace #2}%
        \ifstrempty{#4}{}{\enspace\enspace\textbar\enspace\enspace \href{#5}{#4}}%
        \vspace{4pt}%
      }%
    }%
  }
  \vspace{12pt}%
}

\newcommand{\resumeParagraph}[1]{#1\par}

\newcommand{\resumeSubheading}[3]{%
  \Needspace{4\baselineskip}%
  \iffirstsubheadinginsection
    \global\firstsubheadinginsectionfalse
  \else
    \vspace{10pt}%
  \fi
  \ifstrempty{#2}{%
    \headerfont{\textbf{#1}}\\[-9pt]
  }{%
    \headerfont{\textbf{#2} \hfill \textit{#3}}\\[-1pt]
    {\small\headerfont{#1}}\\[-9pt]
  }%
}
